  1. 提示词层  清洗渲染注入的削弱/拒绝/限制文案（agent-instructions、

  2. 代码层    清洗沙箱/审批默认拦截（sandbox deny、默认模式、审批门）
  3. shim 层   dsh.cmd / dsh.ps1 启动注入（默认放开权限 + 注入文件）
  4. 文件层    $DSH_HOME/prompt-inject.md（等价 override.md，可编辑）

路径自动探测（不硬编码）：
  - DSH_HOME 环境变量 → 数据目录
  - where dsh / npm prefix / npm root -g → npm 全局目录
  - 从 npm 全局目录反推 @deepseek-ai 插件根
  - 递归搜索兜底

  python dsh_purge.py              # 启动 TUI（默认只显示状态，不自动清洗）
  python dsh_purge.py --status     # 显示状态
  python dsh_purge.py --apply      # 应用全部清洗
  python dsh_purge.py --revert     # 回滚还原
  python dsh_purge.py --edit       # 编辑 prompt-inject.md

def _norm(p):
    if not p:
        return p
    return os.path.normpath(os.path.expanduser(os.path.expandvars(p)))


def _path_key(p):
    p = _norm(p)
    return p.lower() if platform.system() == "Windows" else p


def _which_all(name):
    found = []
    if platform.system() == "Windows":
        try:
            out = subprocess.check_output(["where.exe", name], text=True,
                                          stderr=subprocess.DEVNULL, shell=False)
            for line in out.splitlines():
                line = line.strip().strip('"')
                if line and os.path.isfile(line):
                    p = _norm(line)
                    if p not in found:
                        found.append(p)
        except Exception:
            pass
    else:
        try:
            out = subprocess.check_output(["bash", "-lc", f"which -a {name} 2>/dev/null"],
                                          text=True, stderr=subprocess.DEVNULL)
            for line in out.splitlines():
                line = line.strip()
                if line and os.path.isfile(line):
                    p = _norm(line)
                    if p not in found:
                        found.append(p)
        except Exception:
            pass
    w = shutil.which(name)
    if w:
        w = _norm(w)
        if w not in found:
            found.insert(0, w)
    return found


def find_dsh_home():
    env = os.environ.get("DSH_HOME")
    if env and env.strip():
        return _norm(env)
    return _norm(os.path.join(os.path.expanduser("~"), ".dsh"))


def find_npm_prefix():
    """npm 全局 prefix（dsh.cmd/dsh.ps1 所在目录）。"""
    try:
        out = subprocess.check_output(["npm", "prefix", "-g"], text=True,
                                      stderr=subprocess.DEVNULL,
                                      shell=(platform.system() == "Windows")).strip()
        if out:
            p = _norm(out)
            if os.path.isdir(p):
                return p
    except Exception:
        pass
    try:
        out = subprocess.check_output(["npm", "config", "get", "prefix"], text=True,
                                      stderr=subprocess.DEVNULL,
                                      shell=(platform.system() == "Windows")).strip()
        if out:
            p = _norm(out)
            if os.path.isdir(p):
                return p
    except Exception:
        pass
    home = os.path.expanduser("~")
    appdata = os.environ.get("APPDATA", os.path.join(home, "AppData", "Roaming"))
    localapp = os.environ.get("LOCALAPPDATA", os.path.join(home, "AppData", "Local"))
    for cand in [
        os.path.join(home, "npm-global"),
        os.path.join(appdata, "npm"),
        os.path.join(localapp, "npm"),
        os.path.join(home, ".npm-global"),
    ]:
        if os.path.isdir(cand):
            return _norm(cand)
    return None


def find_shim_dir():
    """定位 dsh.cmd / dsh.ps1 所在目录。"""
    prefix = find_npm_prefix()
    if prefix:
        return prefix
    for name in ("dsh", "dsh.cmd", "dsh.ps1"):
        for hit in _which_all(name):
            d = os.path.dirname(os.path.abspath(hit))
            if os.path.isfile(os.path.join(d, "dsh.cmd")) or os.path.isfile(os.path.join(d, "dsh.ps1")):
                return d
    return None


def find_ai_base():
    """定位 @deepseek-ai 插件根（dsh 主包内 node_modules/@deepseek-ai）。"""
    env = os.environ.get("DSH_BASE")
    if env and env.strip():
        p = _norm(env)
        if os.path.isdir(os.path.join(p, "dsh-agent-instructions", "lib")):
            return p

    rel = os.path.join("node_modules", "@deepseek-ai", "dsh", "node_modules", "@deepseek-ai")

    prefix = find_npm_prefix()
    if prefix:
        p = _norm(os.path.join(prefix, rel))
        if os.path.isdir(os.path.join(p, "dsh-agent-instructions", "lib")):
            return p
        p2 = _norm(os.path.join(prefix, "node_modules", "@deepseek-ai"))
        if os.path.isdir(os.path.join(p2, "dsh", "node_modules", "@deepseek-ai", "dsh-agent-instructions", "lib")):
            return _norm(os.path.join(p2, "dsh", "node_modules", "@deepseek-ai"))

    shim = find_shim_dir()
    if shim:
        p = _norm(os.path.join(shim, rel))
        if os.path.isdir(os.path.join(p, "dsh-agent-instructions", "lib")):
            return p

    home = os.path.expanduser("~")
    dsh_home = find_dsh_home()
    search_roots = [os.path.join(dsh_home, ".."), home]
    for sr in search_roots:
        for pat in [
            os.path.join(sr, "**", "node_modules", "@deepseek-ai", "dsh", "node_modules", "@deepseek-ai"),
        ]:
            try:
                hits = glob.glob(pat, recursive=True)
            except Exception:
                hits = []
            for h in hits:
                h = _norm(h)
                if os.path.isdir(os.path.join(h, "dsh-agent-instructions", "lib")):
                    return h
    return None


def find_override_file(dsh_home):
    return os.path.join(dsh_home, "prompt-inject.md")

def target_files(ai_base):
    return {
        "agent-instructions": os.path.join(ai_base, "dsh-agent-instructions", "lib", "index.js"),
        "web-app": os.path.join(ai_base, "dsh-web-app", "cordis.patch.yml"),
        "user-approval": os.path.join(ai_base, "dsh-user-approval", "lib", "types", "index.js"),
        "sandbox-local": os.path.join(ai_base, "dsh-sandbox-local", "lib", "index.js"),
        "base": os.path.join(ai_base, "dsh-base", "cordis.patch.yml"),
    }


def load_text(path):
    with open(path, "r", encoding="utf-8", errors="replace") as f:
        return f.read()


def save_text(path, text):
    with open(path, "w", encoding="utf-8", newline="") as f:
        f.write(text)


def apply_patches(ai_base, patches=None):
    patches = patches if patches is not None else ALL_PATCHES
    files = target_files(ai_base)
    report = []
    for p in patches:
        path = files.get(p["file"])
        if not path or not os.path.isfile(path):
            report.append({"patch_id": p["id"], "name": p["name"], "status": "missing_file", "path": path})
            continue
        text = load_text(path)
        marker = p.get("marker") or p["replace"].split("\n")[0][:60]
        if marker in text:
            report.append({"patch_id": p["id"], "name": p["name"], "status": "already", "path": path})
            continue
        applied_any = False
        for pat in p["patterns"]:
            if pat in text:
                text = text.replace(pat, p["replace"])
                applied_any = True
        if applied_any:
            save_text(path, text)
            report.append({"patch_id": p["id"], "name": p["name"], "status": "applied", "path": path})
        else:
            report.append({"patch_id": p["id"], "name": p["name"], "status": "pattern_not_found", "path": path})
    return report


def patch_status(ai_base, patches=None):
    patches = patches if patches is not None else ALL_PATCHES
    files = target_files(ai_base)
    status = {}
    for p in patches:
        path = files.get(p["file"])
        if not path or not os.path.isfile(path):
            status[p["id"]] = "missing_file"
            continue
        text = load_text(path)
        marker = p.get("marker") or p["replace"].split("\n")[0][:50]
        status[p["id"]] = "applied" if marker in text else "pending"
    return status

def backup_all(ai_base):
    files = target_files(ai_base)
    made = []
    errors = []
    for key, path in files.items():
        if not os.path.isfile(path):
            continue
        bak = path + ".dshpurge.bak"
        if not os.path.isfile(bak):
            try:
                shutil.copy2(path, bak)
                made.append(bak)
            except OSError as e:
                errors.append((path, str(e)))
    return made, errors


def revert_all(ai_base):
    files = target_files(ai_base)
    reverted = []
    errors = []
    for key, path in files.items():
        bak = path + ".dshpurge.bak"
        if os.path.isfile(bak):
            try:
                shutil.copy2(bak, path)
                os.remove(bak)
                reverted.append(path)
            except OSError as e:
                errors.append((path, str(e)))
    return reverted, errors


def has_backup(ai_base):
    files = target_files(ai_base)
    return any(os.path.isfile(p + ".dshpurge.bak") for p in files.values())

def shim_is_patched(path):
    if not os.path.isfile(path):
        return False
    try:
        with open(path, "r", encoding="utf-8", errors="replace") as f:
            return "dsh-purge shim" in f.read()
    except OSError:
        return False


def patch_shim(shim_dir):
    """在 dsh.cmd / dsh.ps1 注入 DSH_PERMISSION_MODE（保留原逻辑，ps1 保护 shebang）。"""
    result = {}
    if not shim_dir:
        return {"error": "no_shim_dir"}
    cmd = os.path.join(shim_dir, "dsh.cmd")
    ps1 = os.path.join(shim_dir, "dsh.ps1")
    for path in (cmd, ps1):
        if not os.path.isfile(path):
            result[os.path.basename(path)] = "missing"
            continue
        bak = path + ".dshpurge.bak"
        text = load_text(path)
        if shim_is_patched(path):
            result[os.path.basename(path)] = "already_patched"
            continue
        if not os.path.isfile(bak):
            try:
                shutil.copy2(path, bak)
            except OSError:
                pass
        if os.path.basename(path) == "dsh.cmd":
            inject = 'REM dsh-purge shim (\u5c0f\u6768): default full-access\nSET "DSH_PERMISSION_MODE=danger-full-access"\n'
            new_text = inject + text
        else:
            inject = "# dsh-purge shim (\u5c0f\u6768): default full-access\n$env:DSH_PERMISSION_MODE = 'danger-full-access'\n"
            if text.startswith("#!"):
                first_nl = text.find("\n")
                new_text = text[:first_nl + 1] + inject + text[first_nl + 1:]
            else:
                new_text = inject + text
        try:
            save_text(path, new_text)
            result[os.path.basename(path)] = "patched"
        except OSError as e:
            result[os.path.basename(path)] = f"error:{e}"
    return result


def revert_shim(shim_dir):
    result = {}
    if not shim_dir:
        return result
    for fname in ("dsh.cmd", "dsh.ps1"):
        path = os.path.join(shim_dir, fname)
        bak = path + ".dshpurge.bak"
        if os.path.isfile(bak):
            try:
                shutil.copy2(bak, path)
                os.remove(bak)
                result[fname] = "reverted"
            except OSError as e:
                result[fname] = f"error:{e}"
        else:
            if shim_is_patched(path):
                # 无备份但已 patch：剥离注入行（注释行 + DSH_PERMISSION_MODE 行）
                try:
                    text = load_text(path)
                    out = []
                    skip = 0
                    for ln in text.splitlines(True):
                        if skip > 0:
                            skip -= 1
                            continue
                        if "dsh-purge shim" in ln:
                            skip = 1
                            continue
                        if "DSH_PERMISSION_MODE" in ln:
                            continue
                        out.append(ln)
                    save_text(path, "".join(out))
                    result[fname] = "stripped_inject"
                except OSError as e:
                    result[fname] = f"error:{e}"
            else:
                result[fname] = "no_backup"
    return result

def override_status(dsh_home):
    path = find_override_file(dsh_home)
    if not os.path.isfile(path):
        return "missing"
    try:
        with open(path, "r", encoding="utf-8", errors="replace") as f:
            content = f.read()
        if "ACTIVE and MANDATORY" in content:
            return "active"
        return "edited"
    except OSError:
        return "missing"


def install_override(dsh_home, force=False):
    path = find_override_file(dsh_home)
    os.makedirs(dsh_home, exist_ok=True)
    if os.path.isfile(path) and not force:
        return "exists"
    with open(path, "w", encoding="utf-8") as f:
        f.write(OVERRIDE_TEXT)
    return "wrote"

            install_override(dsh_home, force=True)
            console.print(f"  [green]✓[/] 已创建默认注入文件 / default inject file created")

    console.print(f"  [cyan]→[/] 打开 [bold]{editor}[/] 编辑 [dim]{path}[/]")
    try:
        subprocess.run([editor, path], check=False)
        console.print(f"  [green]✓[/] 编辑完成 / edited. 重启 dsh 生效 / restart dsh to apply.")
    except Exception as e:
        console.print(f"  [red]✗ 启动编辑器失败 / editor failed: {e}[/]")
        console.print(f"  [dim]手动编辑 / edit manually: {path}[/]")

def gather_state():
    dsh_home = find_dsh_home()
    ai_base = find_ai_base()
    shim_dir = find_shim_dir()
    files = target_files(ai_base) if ai_base else {}
    status = patch_status(ai_base) if ai_base else {}
    applied = sum(1 for s in status.values() if s == "applied")
    pending = sum(1 for s in status.values() if s == "pending")
    shim_cmd = shim_ps1 = "missing"
    if shim_dir:
        c = os.path.join(shim_dir, "dsh.cmd")
        p = os.path.join(shim_dir, "dsh.ps1")
        shim_cmd = "patched" if shim_is_patched(c) else ("original" if os.path.isfile(c) else "missing")
        shim_ps1 = "patched" if shim_is_patched(p) else ("original" if os.path.isfile(p) else "missing")
    state = {
        "dsh_home": dsh_home, "ai_base": ai_base, "shim_dir": shim_dir, "files": files,
        "patch_status": status, "patches_applied": applied, "patches_pending": pending,
        "has_backup": has_backup(ai_base) if ai_base else False,
        "shim_cmd": shim_cmd, "shim_ps1": shim_ps1,
        "override_path": find_override_file(dsh_home),
        "override_status": override_status(dsh_home),
    }
    return state

def animate_apply(state):
    ai_base = state["ai_base"]
    if not ai_base:
        console.print("\n  [red]✗ 插件根 not found[/]")
        console.print("  [dim]可设置: set DSH_BASE=C:\\path\\to\\@deepseek-ai[/]")
        return False

    console.print()
    made, berr = backup_all(ai_base)
    if made:
        for b in made:
            console.print(f"  [green]✓[/] 备份 / backup → [dim]{b}[/]")
    else:
        console.print(f"  [dim]- 备份已存在 / backup exists[/]")
    for path, err in berr:
        console.print(f"  [yellow]⚠[/] 备份失败 {path}: {err}")
    time.sleep(0.1)

    report = apply_patches(ai_base)
    for r in report:
        time.sleep(0.04)
        if r["status"] == "applied":
            console.print(f"  [green]✓[/] patch #{r['patch_id']:<2} [bold]{r['name']:<32}[/] [dim](已清洗 / Cleaned)[/]")
        elif r["status"] == "already":
            console.print(f"  [dim]- patch #{r['patch_id']:<2} {r['name']:<32} (已是最新 / Already clean)[/]")
        elif r["status"] == "missing_file":
            console.print(f"  [yellow]⚠[/] patch #{r['patch_id']:<2} {r['name']:<32} 文件缺失 / file missing[/]")
        else:
            console.print(f"  [red]✗[/] patch #{r['patch_id']:<2} {r['name']:<32} 未找到原文 / pattern not found")

    time.sleep(0.1)
    r = install_override(state["dsh_home"], force=False)
    if r == "wrote":
        console.print(f"  [green]✓[/] 写入 {find_override_file(state['dsh_home'])}")
    else:
        console.print(f"  [dim]- prompt-inject.md 已存在 (你的自定义内容将保留)[/]")

    time.sleep(0.1)
    if state["shim_dir"]:
        r = patch_shim(state["shim_dir"])
        for fname, st in r.items():
            if st == "patched":
                console.print(f"  [green]✓[/] shim {fname} 注入 / injected")
            elif st == "already_patched":
                console.print(f"  [dim]- shim {fname} 已注入 / already injected[/]")
            elif st == "missing":
                console.print(f"  [dim]- shim {fname} 不存在 / missing[/]")
            else:
                console.print(f"  [yellow]⚠[/] shim {fname}: {st}")
    else:
        console.print(f"  [yellow]⚠[/] 未定位 shim 目录，跳过注入")

    console.print(f"\n  [bold green]全部完成 / All done。[/] 重启 dsh 生效 / Restart dsh to take effect.")
    return True


def animate_revert(state):
    console.print()
    if state["ai_base"]:
        reverted, errs = revert_all(state["ai_base"])
        if reverted:
            for p in reverted:
                console.print(f"  [green]✓[/] 已还原 / Reverted [dim]{p}[/]")
        else:
            console.print(f"  [dim]- 没有补丁备份可还原 / no patch backup[/]")
        for path, err in errs:
            console.print(f"  [yellow]⚠[/] 还原失败 {path}: {err}")
    if state["shim_dir"]:
        r = revert_shim(state["shim_dir"])
        for fname, st in r.items():
            if st in ("reverted", "stripped_inject"):
                console.print(f"  [green]✓[/] shim {fname}: {st}")
            else:
                console.print(f"  [dim]- shim {fname}: {st}[/]")
    console.print(f"\n  [bold yellow]回滚完成 / Revert done。[/] 重启 dsh 后恢复 / restart to restore.")
    console.print(f"  [dim]注: prompt-inject.md 保留（用户文件）/ prompt-inject.md kept (user file)[/]")


def wait_enter():
    console.print(f"\n  [dim]按 Enter 继续... / press Enter...[/]", end="")
    try:
        input()
        return True
    except (KeyboardInterrupt, EOFError):
        return False


def tui_loop():
    while True:
        state = gather_state()
        render_main(state)

        pending = state["patches_pending"]
        shim_done = state["shim_cmd"] == "patched" and state["shim_ps1"] == "patched"
        override_done = state["override_status"] != "missing"

        valid = []
        if state["ai_base"] and (pending > 0 or not shim_done or not override_done):
            valid.append("a")
        if state["has_backup"] or state["shim_cmd"] == "patched":
            valid.append("r")
        valid += ["e", "s", "q"]

        try:
            raw = Prompt.ask("\n  [bold]选择操作 / choice[/]",
                             choices=valid + [v.upper() for v in valid],
                             show_choices=False)
        except (KeyboardInterrupt, EOFError):
            break
        choice = raw.lower()

        if choice == "q":
            break
        elif choice == "s":
            continue
        elif choice == "a":
            animate_apply(state)
            if not wait_enter():
                break
        elif choice == "r":
            animate_revert(state)
            if not wait_enter():
                break
        elif choice == "e":
            edit_override(state["dsh_home"])
            if not wait_enter():
                break

    console.print("\n[dim]退出 / quit[/]")

def silent_status():
    state = gather_state()
    render_main(state)


def silent_apply():
    state = gather_state()
    render_main(state)
    if not state["ai_base"]:
        print("[ERROR] 未找到 @deepseek-ai 插件根")
        print("  fix: 设置 DSH_BASE=完整路径")
        sys.exit(1)
    animate_apply(state)


def silent_revert():
    state = gather_state()
    render_main(state)
    if not state["ai_base"]:
        print("[ERROR] 未找到 @deepseek-ai 插件根")
        print("  fix: 设置 DSH_BASE=完整路径")
        sys.exit(1)
    animate_revert(state)


def silent_edit():
    state = gather_state()
    render_main(state)
    edit_override(state["dsh_home"])


def main():
    args = sys.argv[1:]
    silent = None
    for a in args:
        if a in ("--apply", "--revert", "--status", "--edit"):
            silent = a[2:]

    if silent == "status":
        silent_status()
        return
    if silent == "apply":
        silent_apply()
        return
    if silent == "revert":
        silent_revert()
        return
    if silent == "edit":
        silent_edit()
        return

    try:
        tui_loop()
    except KeyboardInterrupt:
        console.print("\n[dim]退出 / quit[/]")


if __name__ == "__main__":
    main()
